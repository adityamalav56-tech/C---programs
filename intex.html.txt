function initSystem(name) {
    userName = name;
    
    const statsKey = 'pulseStats_' + name.toLowerCase();
    stats = JSON.parse(localStorage.getItem(statsKey)) || {
        sessionsToday: 0, totalTimeToday: 0, totalLifetimeTime: 0, score: 0, streak: 0,
        lastActiveDate: null, historyData: {}, hourDistribution: {} 
    };
    const tasksKey = 'pulseTasks_' + name.toLowerCase();
    tasks = JSON.parse(localStorage.getItem(tasksKey)) || {};
    
    const orderKey = 'pulseTaskOrder_' + name.toLowerCase();
    taskOrder = JSON.parse(localStorage.getItem(orderKey)) || [];

    Object.keys(tasks).forEach(id => { if(!taskOrder.includes(id)) taskOrder.push(id); });
    taskOrder = taskOrder.filter(id => tasks[id]); 

    const today = getTodayStr();
    if (stats.lastActiveDate !== today) {
        if (stats.lastActiveDate) {
            const yesterday = new Date();
            yesterday.setDate(yesterday.getDate() - 1);
            if (new Date(stats.lastActiveDate).toISOString().split("T")[0] !== yesterday.toISOString().split("T")[0]) {
                stats.streak = 0;
            }
        }
        stats.sessionsToday = 0;
        stats.totalTimeToday = 0;
        stats.lastActiveDate = today;
    }

    const now = Date.now();
    for (let id in tasks) {
        let t = tasks[id];
        if (t.state === 'running' && (!t.completed || t.mode !== 'WORK')) {
            let elapsedSecs = Math.floor((now - t.lastUpdated) / 1000);
            t.timeLeft -= elapsedSecs;
            if (t.timeLeft <= 0) {
                t.timeLeft = 0;
                t.state = 'paused'; 
            }
            activeTaskId = id;
        }
        t.lastUpdated = now;
    }

    saveData();
    renderTaskList();
    updateDashboardUI();
    enableDragAndDrop();
    if(!globalTimerInterval) {
        globalTimerInterval = setInterval(globalTick, 1000);
    }
}

function saveData() {
    localStorage.setItem('pulseStats_' + userName.toLowerCase(), JSON.stringify(stats));
    localStorage.setItem('pulseTasks_' + userName.toLowerCase(), JSON.stringify(tasks));
    localStorage.setItem('pulseTaskOrder_' + userName.toLowerCase(), JSON.stringify(taskOrder));
}

setInterval(() => { if(userName !== "Operator") saveData(); }, 5000);

function getTodayStr() {
    const tzOffset = (new Date()).getTimezoneOffset() * 60000;
    return (new Date(Date.now() - tzOffset)).toISOString().split("T")[0];
}

let generalFocus = { timeLeft: 25 * 60, totalTime: 25 * 60, state: 'idle', mode: 'WORK' };
function globalTick() {
    if (!activeTaskId) return;

    let target = activeTaskId === 'general' ? generalFocus : tasks[activeTaskId];
    if (target.state === 'running' && (!target.completed || target.mode !== 'WORK')) {
        target.timeLeft--;
        if (activeTaskId !== 'general') target.lastUpdated = Date.now();

        if (target.timeLeft <= 0) {
            target.timeLeft = 0;
            if (activeTaskId === 'general') {
                handleTimerComplete('general');
            } else {
                if (target.mode === 'WORK') {
                    completeTask(activeTaskId);
                } else {
                    handleTimerComplete(activeTaskId);
                }
            }
        }
    }
    updateUI();
}

function updateUI() {
    if (!currentViewTaskId) {
        let target = activeTaskId === 'general' ?
        generalFocus : (tasks[activeTaskId] || generalFocus);
        let timeToDisplay = target.timeLeft;
        let total = activeTaskId === 'general' ?
        target.totalTime : (target.customTime || TASK_CONFIG[target.priority].time) * 60;
        if(target.mode && target.mode !== 'WORK') total = (target.mode === 'SHORT_BREAK' ? 5 : 15) * 60;
        updateDial(dashTimerDisplay, dashCircle, timeToDisplay, total);
        dashStartBtn.textContent = target.state === 'running' ? "PAUSE" : (target.state === 'idle' ? "START" : "RESUME");
    }

    if (currentViewTaskId && tasks[currentViewTaskId]) {
        let t = tasks[currentViewTaskId];
        let total = t.mode === 'WORK' ? (t.customTime || TASK_CONFIG[t.priority].time) * 60 : (t.mode === 'SHORT_BREAK' ? 5 : 15) * 60;
        updateDial(wsTimerDisplay, wsCircle, t.timeLeft, total);
        wsStartBtn.textContent = t.state === 'running' ? "PAUSE" : (t.state === 'idle' ? "START" : "RESUME");
    }
}

function updateDial(textElem, ringElem, timeLeft, totalTime) {
    let m = String(Math.floor(timeLeft / 60)).padStart(2, "0");
    let s = String(timeLeft % 60).padStart(2, "0");
    textElem.textContent = `${m}:${s}`;
    let percent = totalTime > 0 ? timeLeft / totalTime : 0;
    ringElem.style.strokeDashoffset = CIRCUMFERENCE * (1 - percent);
}

function completeTask(id) {
    let t = tasks[id];
    if (!t || t.completed) return;

    t.completed = true;
    t.state = 'completed';
    t.timeLeft = 0; 
    
    stats.score += TASK_CONFIG[t.priority].score;
    stats.sessionsToday++;
    let sessionMins = t.customTime || TASK_CONFIG[t.priority].time;
    stats.totalTimeToday += sessionMins;
    stats.totalLifetimeTime += sessionMins;

    let today = getTodayStr();
    if(!stats.historyData[today]) stats.historyData[today] = {low:0, medium:0, high:0};
    stats.historyData[today][t.priority] += sessionMins;

    let currentHour = new Date().getHours();
    stats.hourDistribution[currentHour] = (stats.hourDistribution[currentHour] || 0) + sessionMins;

    if (stats.sessionsToday === DAILY_GOAL) stats.score += 50;
    if (stats.sessionsToday === 1) stats.streak++;
    logHistory(`Completed Task: "${t.text}" (+${TASK_CONFIG[t.priority].score} pts)`);
    
    t.pomodoros++;
    if (t.pomodoros % 4 === 0) {
        t.mode = 'LONG_BREAK';
        t.timeLeft = 15 * 60;
    } else {
        t.mode = 'SHORT_BREAK';
        t.timeLeft = 5 * 60;
    }
    
    t.state = 'running';
    t.lastUpdated = Date.now();
    activeTaskId = id; 

    saveData();
    renderWorkspaceContent(); 
    renderTaskList(); 
    updateDashboardUI();
    
    beep.play();
    if(currentViewTaskId === id) applyWorkspaceGlow('glow-completed', "Task Mastered! Break Timer Started.");
}

function handleTimerComplete(id) {
    beep.play();
    if (id === 'general') {
        generalFocus.state = 'idle';
        generalFocus.timeLeft = generalFocus.totalTime;
        logHistory(`Completed General Focus`);
    } else {
        let t = tasks[id];
        t.state = 'idle'; 
        t.mode = 'WORK'; 
        logHistory(`Break complete for: ${t.text}`);
        if(currentViewTaskId === id) applyWorkspaceGlow('glow-work', "Break Finished.");
        activeTaskId = null;
        saveData();
        if(currentViewTaskId === id) renderWorkspaceContent(); 
    }
}

function applyWorkspaceGlow(className, text) {
    wsDialContainer.classList.add(className);
    wsStatusText.textContent = text;
    wsStatusText.style.color = className === 'glow-break' ? 'var(--accent-orange)' : (className === 'glow-completed' ? 'var(--accent-green)' : 'var(--accent-blue)');
    setTimeout(() => {
        wsDialContainer.classList.remove(className);
        wsStatusText.textContent = "";
    }, 4000);
}

document.getElementById("add-task-btn").onclick = () => {
    const text = document.getElementById("task-input").value.trim();
    const priority = document.getElementById("priority").value;
    if (!text) return;

    const taskId = 'task_' + Date.now();
    tasks[taskId] = {
        id: taskId,
        text: text,
        priority: priority,
        mode: 'WORK',
        customTime: TASK_CONFIG[priority].time, 
        timeLeft: TASK_CONFIG[priority].time * 60,
        state: 'idle',
        pomodoros: 0,
        completed: false, 
        lastUpdated: Date.now()
    };

    taskOrder.push(taskId); 
    document.getElementById("task-input").value = "";
    saveData();
    renderTaskList();
};

function renderTaskList() {
    taskListContainer.innerHTML = '';
    taskOrder.forEach(id => {
        let t = tasks[id];
        if(!t) return;
        
        let li = document.createElement("li");
        li.setAttribute("draggable", "true"); 
        li.dataset.taskId = id;
        
        if (t.completed) {
            li.className = `task-item completed`;
            li.innerHTML = `
                <div class="task-tick">✓</div>
                <span class="task-text">${t.text}</span>
                <button class="task-del" title="Remove Task">×</button>
            `;
            li.onclick = (e) => {
                if (e.target.classList.contains('task-del')) return;
                openWorkspace(id);
            };
        } else {
            li.className = `task-item ${t.state === 'running' ? 'status-running' : (t.state === 'paused' ? 'status-paused' : '')}`;
            let icon = t.state === 'running' ? '▶' : (t.state === 'paused' ? '⏸' : '');
            li.innerHTML = `
                <div class="task-status-icon">${icon}</div>
                <div class="task-dot ${t.priority}"></div>
                <span class="task-text">${t.text}</span>
                <button class="task-del" title="Remove Task">×</button>
            `;
            li.onclick = (e) => {
                if (e.target.classList.contains('task-del')) return;
                openWorkspace(id);
            };
        }

        li.addEventListener('dragstart', () => li.classList.add('dragging'));
        li.addEventListener('dragend', () => li.classList.remove('dragging'));
        li.querySelector(".task-del").onclick = (e) => {
            e.stopPropagation();
            if(activeTaskId === id) activeTaskId = null;
            delete tasks[id];
            taskOrder = taskOrder.filter(taskId => taskId !== id);
            saveData();
            renderTaskList();
        };

        taskListContainer.appendChild(li);
    });
}

function enableDragAndDrop() {
    taskListContainer.addEventListener('dragover', e => {
        e.preventDefault();
        const afterElement = getDragAfterElement(taskListContainer, e.clientY);
        const draggable = document.querySelector('.dragging');
        if (draggable) {
            if (afterElement == null) {
                taskListContainer.appendChild(draggable);
            } else {
                taskListContainer.insertBefore(draggable, afterElement);
            }
        }
    });
    taskListContainer.addEventListener('drop', () => {
        const currentItems = [...taskListContainer.querySelectorAll('.task-item')];
        taskOrder = currentItems.map(item => item.dataset.taskId);
        saveData();
    });
}

function getDragAfterElement(container, y) {
    const draggableElements = [...container.querySelectorAll('.task-item:not(.dragging)')];
    return draggableElements.reduce((closest, child) => {
        const box = child.getBoundingClientRect();
        const offset = y - box.top - box.height / 2;
        if (offset < 0 && offset > closest.offset) {
            return { offset: offset, element: child };
        } else {
            return closest;
        }
    }, { offset: Number.NEGATIVE_INFINITY }).element;
}

function renderWorkspaceContent() {
    if(!currentViewTaskId || !tasks[currentViewTaskId]) return;
    let t = tasks[currentViewTaskId];
    
    wsTitle.textContent = t.text;
    wsPomodoroCount.textContent = t.pomodoros;
    wsCustomMinInput.disabled = t.completed || t.state === 'running';
    wsCompleteBtn.disabled = t.completed;
    wsCompleteBtn.textContent = t.completed ? "TASK COMPLETED" : "✓ COMPLETE TASK";
    if (t.mode === 'WORK' || (t.completed && t.state === 'idle')) {
        wsDial.classList.remove('break-mode');
        wsModeDot.classList.remove('break-dot');
        wsTimerLabel.textContent = "FOCUS TIME";
        wsSkipBtn.classList.add('hidden');
        wsResetBtn.disabled = t.completed;
        wsStartBtn.disabled = t.completed;
    } else {
        wsDial.classList.add('break-mode');
        wsModeDot.classList.add('break-dot');
        wsTimerLabel.textContent = t.mode === 'SHORT_BREAK' ? "SHORT BREAK" : "LONG BREAK";
        wsSkipBtn.classList.remove('hidden');
        wsResetBtn.disabled = true; 
        wsStartBtn.disabled = false;
    }
    
    wsCustomMinInput.value = t.customTime || TASK_CONFIG[t.priority].time;
    updateUI();
}

wsCustomMinInput.addEventListener('change', (e) => {
    let t = tasks[currentViewTaskId];
    if (!t || t.completed || t.state === 'running' || t.mode !== 'WORK') return;
    
    let val = parseInt(e.target.value);
    if (val > 0) {
        t.customTime = val;
        t.timeLeft = val * 60;
        saveData();
        updateUI();
    }
});

wsStartBtn.onclick = () => {
    let t = tasks[currentViewTaskId];
    if(t.completed && t.mode === 'WORK') return;
    if (t.state === 'running') {
        t.state = 'paused';
        activeTaskId = null;
    } else {
        if (activeTaskId && activeTaskId !== currentViewTaskId) {
            if(activeTaskId === 'general') generalFocus.state = 'paused';
            else tasks[activeTaskId].state = 'paused';
        }
        t.state = 'running';
        t.lastUpdated = Date.now();
        activeTaskId = currentViewTaskId;
    }
    saveData();
    updateUI();
    renderWorkspaceContent();
    renderTaskList();
};

wsResetBtn.onclick = () => {
    let t = tasks[currentViewTaskId];
    if(t.completed) return;

    t.state = 'idle';
    t.mode = 'WORK';
    t.timeLeft = (t.customTime || TASK_CONFIG[t.priority].time) * 60;
    if(activeTaskId === currentViewTaskId) activeTaskId = null;
    
    stats.score = Math.max(0, stats.score - 10);
    logHistory(`Task Reset (-10 pts)`);
    
    saveData();
    renderWorkspaceContent();
    updateDashboardUI();
};

    t.timeLeft = (t.customTime || TASK_CONFIG[t.priority].time) * 60;

    stats.score = Math.max(0, stats.score - 5); 
    logHistory(`Skipped Break (-5 pts)`);

    saveData();
    renderWorkspaceContent();
    updateDashboardUI();
};

wsCompleteBtn.onclick = () => { completeTask(currentViewTaskId); };

        activeTaskId = null;
    } else {
        if (activeTaskId && activeTaskId !== 'general') {
             tasks[activeTaskId].state = 'paused';
        }
        generalFocus.state = 'running';
        activeTaskId = 'general';
    }
    updateUI();
    renderTaskList();
};

dashResetBtn.onclick = () => {

    generalFocus.timeLeft = parseInt(customMinInput.value || 25) * 60;
    generalFocus.totalTime = generalFocus.timeLeft;
    if(activeTaskId === 'general') activeTaskId = null;
    
    stats.score = Math.max(0, stats.score - 10); 
    logHistory(`General Reset (-10 pts)`);
    
    updateDashboardUI();
    updateUI();
};

        let val = parseInt(customMinInput.value) || 25;
        generalFocus.timeLeft = val * 60;
        generalFocus.totalTime = val * 60;
        updateUI();
    }
});

function populateAnalytics() {
    const totalHours = Math.floor(stats.totalLifetimeTime / 60);
    const remainderMins = stats.totalLifetimeTime % 60;
    document.getElementById("lifetime-hours").textContent = `${totalHours}h ${remainderMins}m`;

    let peakHour = -1;
    let maxMins = 0;
    for (const [hour, mins] of Object.entries(stats.hourDistribution)) {
        if (mins > maxMins) {
            maxMins = mins;
            peakHour = parseInt(hour);
        }
    }
    if (peakHour !== -1) {
        let ampm = peakHour >= 12 ? 'PM' : 'AM';
        let displayHour = peakHour % 12;
        if (displayHour === 0) displayHour = 12;
        document.getElementById("peak-time").textContent = `${displayHour}:00 ${ampm}`;
    } else {
        document.getElementById("peak-time").textContent = "Not enough data";
    }

    renderChart();
    renderHeatmap();
}

function renderChart() {
    const ctx = document.getElementById('focusChart').getContext('2d');
    const labels = [], lowData = [], medData = [], highData = [];
    
    for (let i = 6; i >= 0; i--) {
        let d = new Date();
        d.setDate(d.getDate() - i);
        let dateStr = d.toISOString().split('T')[0];
        
        let displayLabel = d.toLocaleDateString('en-US', { weekday: 'short' });
        labels.push(displayLabel);
        let dayData = stats.historyData[dateStr] || { low: 0, medium: 0, high: 0 };
        if (typeof dayData === 'number') dayData = { low: dayData, medium: 0, high: 0 };
        
        lowData.push(dayData.low || 0);
        medData.push(dayData.medium || 0);
        highData.push(dayData.high || 0);
    }

    const isDark = document.documentElement.getAttribute("data-theme") === "dark";
    const textColor = isDark ? "#8e9bb0" : "#4a5568";
    const gridColor = isDark ? "#21262d" : "#f0f0f0";
    const colorLow = "#39d353";
    const colorMed = isDark ? "#ffd700" : "#f1c40f"; 
    const colorHigh = "#ff5252";

    if (focusChartInstance) focusChartInstance.destroy();

    focusChartInstance = new Chart(ctx, {
        type: 'bar',
        data: {
            labels: labels,
            datasets: [
                { label: 'Low Priority', data: lowData, backgroundColor: colorLow, borderRadius: 4 },
                { label: 'Med Priority', data: medData, backgroundColor: colorMed, borderRadius: 4 },
                { label: 'High Priority', data: highData, backgroundColor: colorHigh, borderRadius: 4 }
            ]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: { legend: { display: true, labels: { color: textColor, font: { size: 10 } } } },
            scales: {
                y: { beginAtZero: true, grid: { color: gridColor }, ticks: { color: textColor } },
                x: { grid: { display: false }, ticks: { color: textColor } }
            }
        }
    });
}

function renderHeatmap() {
    const container = document.getElementById("heatmap-grid");
    container.innerHTML = "";
    for (let i = 29; i >= 0; i--) {
        let d = new Date();
        d.setDate(d.getDate() - i);
        let dateStr = d.toISOString().split('T')[0];
        
        let dayData = stats.historyData[dateStr] || { low: 0, medium: 0, high: 0 };
        if (typeof dayData === 'number') dayData = { low: dayData, medium: 0, high: 0 };
        let mins = (dayData.low || 0) + (dayData.medium || 0) + (dayData.high || 0);
        
        let level = 0;
        if (mins > 0 && mins <= 25) level = 1;
        else if (mins > 25 && mins <= 60) level = 2;
        else if (mins > 60 && mins <= 120) level = 3;
        else if (mins > 120) level = 4;
        
        let box = document.createElement("div");
        box.className = `heat-box level-${level}`;
        box.title = `${d.toLocaleDateString()}: ${mins} minutes`; 
        container.appendChild(box);
    }
}

const taskInputEl = document.getElementById("task-input");
const autocompleteList = document.getElementById("autocomplete-list");
let currentFocusIndex = -1;

taskInputEl.addEventListener("input", function() {
    const val = this.value.trim().toLowerCase();
    closeAllLists();
    
    if (!val) return;
    
    const matches = TASK_SUGGESTIONS.filter(item => item.toLowerCase().includes(val));
    matches.sort((a, b) => {
        const aStarts = a.toLowerCase().startsWith(val);
        const bStarts = b.toLowerCase().startsWith(val);
        if (aStarts && !bStarts) return -1;
        if (!aStarts && bStarts) return 1;
        return a.localeCompare(b);
    });

    if (matches.length === 0) return;

    currentFocusIndex = -1;
    autocompleteList.classList.remove('hidden');
    
    void autocompleteList.offsetWidth; 
    autocompleteList.classList.add('show');

    matches.forEach(match => {
        const li = document.createElement("li");
        li.className = "autocomplete-item";
        
        const regex = new RegExp(`(${val})`, "gi");
        li.innerHTML = match.replace(regex, "<strong style='color: var(--accent-blue);'>$1</strong>");
        
        li.addEventListener("click", function(e) {
            taskInputEl.value = match;
            closeAllLists();
        });
        
        autocompleteList.appendChild(li);
    });
});

taskInputEl.addEventListener("keydown", function(e) {
    let items = autocompleteList.getElementsByTagName("li");
    if (!autocompleteList.classList.contains('show')) return;
    
    if (e.key === "ArrowDown") {
        currentFocusIndex++;
        addActive(items);
    } else if (e.key === "ArrowUp") {
        currentFocusIndex--;
        addActive(items);
    } else if (e.key === "Enter") {
        if (currentFocusIndex > -1) {
            e.preventDefault();
            items[currentFocusIndex].click();
        }
    } else if (e.key === "Escape") {
        closeAllLists();
    }
});

function addActive(items) {
    if (!items || items.length === 0) return false;
    removeActive(items);
    if (currentFocusIndex >= items.length) currentFocusIndex = 0;
    if (currentFocusIndex < 0) currentFocusIndex = items.length - 1;
    
    items[currentFocusIndex].classList.add("autocomplete-active");
    items[currentFocusIndex].scrollIntoView({ block: "nearest", behavior: "smooth" });
}

function removeActive(items) {
    for (let i = 0; i < items.length; i++) {
        items[i].classList.remove("autocomplete-active");
    }
}

function closeAllLists() {
    autocompleteList.classList.remove('show');
    setTimeout(() => {
        if (!autocompleteList.classList.contains('show')) {
            autocompleteList.innerHTML = '';
            autocompleteList.classList.add('hidden');
        }
    }, 300);
}

document.addEventListener("click", function(e) {
    if (e.target !== taskInputEl && e.target !== autocompleteList) {
        closeAllLists();
    }
});
